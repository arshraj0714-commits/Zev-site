   167→  // Handle .txt file upload for new stock — parse credentials separated by newlines or commas
   168→  function handleCredFileUpload(e: React.ChangeEvent<HTMLInputElement>) {
   169→    const file = e.target.files?.[0];
   170→    if (!file) return;
   171→    const reader = new FileReader();
   172→    reader.onload = () => {
   173→      const text = reader.result as string;
   174→      const lines = text.split(/\r?\n/).filter((l) => l.trim());
   175→      const parsed: { label: string; value: string }[] = [];
   176→      for (const line of lines) {
   177→        const parts = line.split(",").map((p) => p.trim()).filter(Boolean);
   178→        for (const part of parts) {
   179→          parsed.push({ label: `Item ${parsed.length + 1}`, value: part });
   180→        }
   181→      }
   182→      if (parsed.length > 0) {
   183→        setCreds(parsed);
   184→        toast.success(`Loaded ${parsed.length} credentials from file`);
   185→      } else {
   186→        toast.error("No credentials found in file");
   187→      }
   188→    };
   189→    reader.readAsText(file);
   190→    e.target.value = "";
   191→  }
