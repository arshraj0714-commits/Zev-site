   114→  async function saveProduct() {
   115→    if (!pName || !pDesc) { toast.error("Name and description required"); return; }
   116→    setSavingP(true);
   117→    try {
   118→      const payload = JSON.stringify({
   119→        name: pName, description: pDesc, image: pImage, codeLink: pLink,
   120→        folder: pFolder, type: pType, price: pType === "free" ? 0 : Number(pPrice) || 0,
   121→        tags: pTags, featured: pFeatured,
   122→        fileName: pFile?.name ?? null, fileData: pFile?.data ?? null, fileSize: pFile?.size ?? null,
   123→      });
   124→      // Vercel's serverless function body limit is 4.5MB
   125→      if (payload.length > 4_200_000) {
   126→        toast.error(`Upload too large (${(payload.length / (1024 * 1024)).toFixed(1)}MB). Vercel's limit is 4.5MB. Use a code link (GitHub/Drive) instead of attaching the file, or use a smaller image.`);
   127→        setSavingP(false);
   128→        return;
   129→      }
   130→      const res = await fetch("/api/products", {
   131→        method: "POST",
   132→        headers: { "Content-Type": "application/json" },
   133→        body: payload,
   134→      });
   135→      if (!res.ok) {
   136→        const data = await res.json().catch(() => ({}));
   137→        if (res.status === 413) {
   138→          throw new Error("Upload too large (413). Vercel's body limit is 4.5MB. Use a code link instead of attaching the file, or compress your image.");
   139→        }
   140→        throw new Error(data.error || `Request failed (${res.status})`);
   141→      }
   142→      toast.success("Product added successfully!");
   143→      setPName(""); setPDesc(""); setPImage(null); setPLink(""); setPFolder("");
   144→      setPPrice(""); setPTags(""); setPFeatured(false); setPFile(null);
   145→      qc.invalidateQueries({ queryKey: ["products"] });
   146→    } catch (e) { toast.error((e as Error).message || "Failed to add product"); }
   147→    finally { setSavingP(false); }
   148→  }

   186→  // Handle .txt file upload for new stock — parse credentials separated by newlines or commas
   187→  function handleCredFileUpload(e: React.ChangeEvent<HTMLInputElement>) {
   188→    const file = e.target.files?.[0];
   189→    if (!file) return;
   190→    const reader = new FileReader();
   191→    reader.onload = () => {
   192→      const text = reader.result as string;
   193→      const lines = text.split(/\r?\n/).filter((l) => l.trim());
   194→      const parsed: { label: string; value: string }[] = [];
   195→      for (const line of lines) {
   196→        const parts = line.split(",").map((p) => p.trim()).filter(Boolean);
   197→        for (const part of parts) {
   198→          parsed.push({ label: `Item ${parsed.length + 1}`, value: part });
   199→        }
   200→      }
   201→      if (parsed.length > 0) {
   202→        setCreds(parsed);
   203→        toast.success(`Loaded ${parsed.length} credentials from file`);
   204→      } else {
   205→        toast.error("No credentials found in file");
   206→      }
   207→    };
   208→    reader.readAsText(file);
   209→    e.target.value = "";
   210→  }
